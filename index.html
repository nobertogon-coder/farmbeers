async function processCompleteScreening() { if (medicationsList.length === 0) { alert("Insira ao menos um medicamento para avaliar."); return; } const ageInput = document.getElementById("inpAge").value; const ageVal = parseInt(ageInput) || 70; const payload = { age: ageVal, sector: document.getElementById("selSector").value, fall_risk: document.getElementById("chkFallRisk").checked, conditions: document.getElementById("inpConditions").value.split(/[,]+/).map(c => c.trim()).filter(Boolean), crcl_ml_min: calculatedCrCl, crcl_notes: crclClinicalNote, medications: medicationsList }; try { const res = await fetch("/api/v1/screen", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (!res.ok) { const errData = await res.json().catch(() => ({})); throw new Error(errData.detail || `Erro no servidor (Status ${res.status})`); } const data = await res.json(); lastGeneratedAlerts = data.alerts || []; const box = document.getElementById("screeningReportBox"); const list = document.getElementById("reportAlertsList"); const badge = document.getElementById("reportBadgeSummary"); box.classList.remove("hidden"); badge.innerText = `${data.total_beers_alerts || lastGeneratedAlerts.length} Alertas Identificados`; list.innerHTML = ""; if (lastGeneratedAlerts.length === 0) { list.innerHTML = `
✓ Prescrição validada. Nenhum critério de Beers violado para o perfil clínico informado.
`; } else { let alertsHtml = ""; lastGeneratedAlerts.forEach(a => { const drugName = a.drug || a.drug_name || "Medicamento"; const axisName = a.axis || "AGS Beers 2023"; const recText = a.recommendation || "Reavaliar uso em idosos."; const ratText = a.rationale || "Medicamento potencialmente inapropriado pela diretriz AGS Beers."; const isSn = a.is_sos || false; const snTag = isSn ? 'Regime: SN / QN' : ''; const taperingHtml = a.tapering ? `
Plano Prático de Desprescrição (Tapering):

${a.tapering}

` : ''; alertsHtml += `
⚠ ${drugName} ${snTag} ${axisName}
Conduta Recomendada: ${recText}
Racional Clínico: ${ratText}
${taperingHtml}
`; }); list.innerHTML = alertsHtml; } try { if (window.lucide) lucide.createIcons(); } catch (iconErr) { console.warn("Erro ao carregar ícones:", iconErr); } box.scrollIntoView({ behavior: "smooth" }); } catch (err) { alert("Falha ao processar triagem: " + err.message); } }
